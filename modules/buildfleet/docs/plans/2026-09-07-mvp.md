**Goal:** Implement the MVP defined in DESIGN.md §12.1: Scheduler + RuntimePool + RuntimeProfile + DockerProvider/K8sProvider + LocalStore + BuildKit runtime + credential table (session-based) + static registry config, exposed as a Go SDK.

**Architecture:** Single-process library. A facade (`fleet`) orchestrates: Scheduler picks a warm runtime from an in-memory Pool (or asks a Provider to provision one), Builder connects to the runtime's buildkit gRPC and Solves with credentials passed per-session, build contexts are exchanged by digest reference via LocalStore. Docker/K8s are pluggable providers; registry connection is static config rendered into buildkitd.toml.

**Tech Stack:** Go 1.27, `github.com/moby/buildkit/client` (buildkit gRPC client + session auth), `github.com/docker/docker/client` (DockerProvider), `k8s.io/client-go` (K8sProvider). Tests: stdlib only, table-driven, no testify.

**Read DESIGN.md first.** Two invariants from the design that tests must lock in:
1. Instances hold **no credentials**; everything per-request travels via Solve session.
2. Registry protocol config is **static per-instance** (rendered at provision time); request references an unknown registry → business-class error, no probing.

```go
// Package resource provides build resource quantities and helpers.
package resource

import (
	"fmt"
	"strconv"
	"strings"
)

type Bytes int64

const (
	Kilo Bytes = 1024
	Mega       = 1024 * Kilo
	Giga       = 1024 * Mega
)

var units = map[string]Bytes{"Ki": Kilo, "Mi": Mega, "Gi": Giga, "Ti": 1024 * Giga}

// ParseQuantity parses "<n>" (bytes) or "<n><Ki|Mi|Gi|Ti>".
func ParseQuantity(s string) (Bytes, error) {
	for suf, mult := range units {
		if strings.HasSuffix(s, suf) {
			n, err := strconv.ParseFloat(strings.TrimSuffix(s, suf), 64)
			if err != nil {
				return 0, fmt.Errorf("parse quantity %q: %w", s, err)
			}
			return Bytes(n * float64(mult)), nil
		}
	}
	n, err := strconv.ParseInt(s, 10, 64)
	if err != nil {
		return 0, fmt.Errorf("parse quantity %q: unsupported format", s)
	}
	return Bytes(n), nil
}

func (b Bytes) String() string {
	switch {
	case b >= Giga && b%Giga == 0:
		return fmt.Sprintf("%dGi", b/Giga)
	case b >= Mega && b%Mega == 0:
		return fmt.Sprintf("%dMi", b/Mega)
	default:
		return strconv.FormatInt(int64(b), 10)
	}
}

type Resources struct {
	CPU    float64
	Memory Bytes
}

// Fits reports whether r can hold req (zero-value fields are unbounded).
func (r Resources) Fits(req Resources) bool {
	if req.CPU > r.CPU {
		return false
	}
	if req.Memory > r.Memory {
		return false
	}
	return true
}

func (r Resources) Add(o Resources) Resources {
	return Resources{CPU: r.CPU + o.CPU, Memory: r.Memory + o.Memory}
}

func (r Resources) Sub(o Resources) Resources {
	return Resources{CPU: r.CPU - o.CPU, Memory: r.Memory - o.Memory}
}
```

```go
package resource

import (
	"sort"
	"time"
)

type Entry struct {
	Name        string
	Resources   Resources
	StartupCost time.Duration
}

// Profile maps profile names to resource tiers. Fit returns the smallest
// entry whose Resources hold req (zero-value req fits the smallest tier).
func (p Profile) Fit(req Resources) (Entry, bool) {
	entries := make([]Entry, 0, len(p))
	for _, e := range p {
		entries = append(entries, e)
	}
	sort.Slice(entries, func(i, j int) bool {
		a, b := entries[i].Resources, entries[j].Resources
		if a.CPU != b.CPU {
			return a.CPU < b.CPU
		}
		return a.Memory < b.Memory
	})
	for _, e := range entries {
		if e.Resources.Fits(req) {
			return e, true
		}
	}
	return Entry{}, false
}

// DefaultProfile matches DESIGN.md §2.
func DefaultProfile() Profile {
	return Profile{
		"small":  Entry{Name: "small", Resources: Resources{CPU: 1, Memory: 2 * Giga}, StartupCost: 5 * time.Second},
		"medium": Entry{Name: "medium", Resources: Resources{CPU: 2, Memory: 4 * Giga}, StartupCost: 5 * time.Second},
		"large":  Entry{Name: "large", Resources: Resources{CPU: 4, Memory: 8 * Giga}, StartupCost: 8 * time.Second},
	}
}
```

```go
// Package runtime models pooled buildkitd instances as first-class
// resources, independent of the docker/k8s supply layer.
package runtime

import (
	"sync"
	"time"

	"github.com/funcx27/kite/modules/buildfleet/resource"
)

type State string

const (
	Warm      State = "warm"
	Busy      State = "busy"
	Draining  State = "draining"
	Failed    State = "failed"
)

type Spec struct {
	ID          string
	Resources   resource.Resources
	Platforms   []string
	Capabilities []string
	StartupCost time.Duration
}

type Runtime struct {
	ID           string
	Spec         Spec
	Addr         string // buildkitd gRPC address
	State        State
	ConfigGen    int   // buildkitd.toml generation this instance was rendered from
	BuildkitVer  string

	lastReleased time.Time
}

// Lease is an exclusive rental on a Runtime. Release returns it to the pool
// (warm); Fail marks it failed (not reusable without provider recovery).
type Lease struct {
	rt  *Runtime
	mu  *sync.Mutex
	rel bool
}

func (l *Lease) Runtime() *Runtime { return l.rt }

func (l *Lease) Release() {
	l.mu.Lock()
	defer l.mu.Unlock()
	if !l.rel {
		l.rel = true
		l.rt.State = Warm
		l.rt.lastReleased = time.Now()
	}
}

func (l *Lease) Fail() {
	l.mu.Lock()
	defer l.mu.Unlock()
	if !l.rel {
		l.rel = true
		l.rt.State = Failed
	}
}
```

`runtime/pool.go`:

```go
package runtime

import (
	"context"
	"errors"
	"sort"
	"sync"
	"time"

	"github.com/funcx27/kite/modules/buildfleet/resource"
)

var ErrNoMatch = errors.New("runtime: no matching runtime and no capacity to provision")

// Pool is the in-memory runtime registry. Pure memory state: on restart,
// call ReconcileDrain before serving traffic (DESIGN.md §5.1).
type Pool struct {
	mu       sync.Mutex
	runtimes map[string]*Runtime
	provider Provider
}

func NewPool(p Provider) *Pool {
	return &Pool{runtimes: map[string]*Runtime{}, provider: p}
}

// Register adds a runtime to the pool (called by providers after readiness).
func (p *Pool) Register(r *Runtime) {
	p.mu.Lock()
	defer p.mu.Unlock()
	if r.State == "" {
		r.State = Warm
	}
	p.runtimes[r.ID] = r
}

// Acquire returns an exclusive lease on a warm runtime matching spec, or
// provisions one via the provider. Zero-value spec fields match anything.
func (p *Pool) Acquire(ctx context.Context, spec Spec) (*Lease, error) {
	p.mu.Lock()
	r := p.pickLocked(spec)
	if r != nil {
		r.State = Busy
		p.mu.Unlock()
		return &Lease{rt: r, mu: &p.mu}, nil
	}
	p.mu.Unlock()

	// No match: provision out of the pool lock (provisioning is slow).
	// Capacity check is advisory; the real adjudication happens at the
	// provider (k8s apiserver quota, docker run failure).
	if !p.provider.Capacity().Fits(spec.Resources) {
		return nil, ErrNoMatch
	}
	rt, err := p.provider.Provision(ctx, spec)
	if err != nil {
		return nil, err
	}
	p.Register(rt)
	p.mu.Lock()
	rt.State = Busy
	p.mu.Unlock()
	return &Lease{rt: rt, mu: &p.mu}, nil
}

// pickLocked applies hard filters then orders by startup penalty and
// most-recently-released (LRU keep-alive), with random tie-breaking.
// (DESIGN.md §3: no bin-packing score — warm instances are idle instances.)
func (p *Pool) pickLocked(spec Spec) *Runtime {
	var cands []*Runtime
	for _, r := range p.runtimes {
		if r.State != Warm {
			continue
		}
		if !r.Spec.Resources.Fits(spec.Resources) {
			continue
		}
		if !platformsFit(r.Spec.Platforms, spec.Platforms) {
			continue
		}
		if !capsFit(r.Spec.Capabilities, spec.Capabilities) {
			continue
		}
		cands = append(cands, r)
	}
	if len(cands) == 0 {
		return nil
	}
	sort.Slice(cands, func(i, j int) bool {
		if cands[i].Spec.StartupCost != cands[j].Spec.StartupCost {
			return cands[i].Spec.StartupCost < cands[j].Spec.StartupCost
		}
		return cands[i].lastReleased.Before(cands[j].lastReleased)
	})
	return cands[0]
}

func platformsFit(have, want []string) bool {
	return subset(have, want)
}

func capsFit(have, want []string) bool {
	return subset(have, want)
}

func subset(have, want []string) bool {
	if len(want) == 0 {
		return true
	}
	m := map[string]bool{}
	for _, h := range have {
		m[h] = true
	}
	for _, w := range want {
		if !m[w] {
			return false
		}
	}
	return true
}

// ReconcileDrain marks every runtime draining: used after process restart
// when lease ownership is unrecoverable (in-flight builds are not adopted;
// callers recover via the retry chain).
func (p *Pool) ReconcileDrain() {
	p.mu.Lock()
	defer p.mu.Unlock()
	for _, r := range p.runtimes {
		if r.State == Busy || r.State == Warm {
			r.State = Draining
		}
	}
}

// Drain empties draining/failed runtimes by disposing them via the provider.
func (p *Pool) Drain(ctx context.Context) error {
	p.mu.Lock()
	var ids []string
	for id, r := range p.runtimes {
		if r.State == Draining || r.State == Failed {
			ids = append(ids, id)
			delete(p.runtimes, id)
		}
	}
	p.mu.Unlock()
	for _, id := range ids {
		if err := p.provider.Dispose(ctx, id); err != nil {
			return err
		}
	}
	return nil
}

// Stats snapshots pool state for observability (DESIGN.md §9).
type Stat struct {
	ID    string
	State State
	Addr  string
}

func (p *Pool) Stats() []Stat {
	p.mu.Lock()
	defer p.mu.Unlock()
	out := make([]Stat, 0, len(p.runtimes))
	for _, r := range p.runtimes {
		out = append(out, Stat{ID: r.ID, State: r.State, Addr: r.Addr})
	}
	return out
}

func (p *Pool) forEach(fn func(*Runtime)) {
	p.mu.Lock()
	defer p.mu.Unlock()
	for _, r := range p.runtimes {
		fn(r)
	}
}

var _ = resource.Resources{} // keep import until Capacity types land in providers
```

```go
	"encoding/json"

// LocalStore is a filesystem-backed Store: blobs under <root>/blobs/<aa>/<rest>,
// refcounts under <root>/refs/<digest>.json so they survive restarts.
type LocalStore struct {
	root string
	mu   sync.Mutex
}

func NewLocalStore(root string) *LocalStore {
	return &LocalStore{root: root}
}

func (s *LocalStore) Push(ctx context.Context, r io.Reader) (string, error) {
	tmp, err := os.CreateTemp(s.blobDir(), "ingest-*")
	if err != nil {
		return "", fmt.Errorf("localstore push: create tmp: %w", err)
	}
	defer os.Remove(tmp.Name())

	h := sha256.New()
	if _, err := io.Copy(io.MultiWriter(tmp, h), r); err != nil {
		tmp.Close()
		return "", fmt.Errorf("localstore push: copy: %w", err)
	}
	if err := tmp.Close(); err != nil {
		return "", fmt.Errorf("localstore push: close: %w", err)
	}
	digest := "sha256:" + hex.EncodeToString(h.Sum(nil))
	dst := s.path(digest)
	if _, err := os.Stat(dst); err == nil {
		return digest, nil // dedup
	}
	if err := os.MkdirAll(filepath.Dir(dst), 0o755); err != nil {
		return "", fmt.Errorf("localstore push: mkdir: %w", err)
	}
	if err := os.Rename(tmp.Name(), dst); err != nil {
		return "", fmt.Errorf("localstore push: rename: %w", err)
	}
	return digest, nil
}

func (s *LocalStore) Pull(ctx context.Context, digest string) (io.ReadCloser, error) {
	f, err := os.Open(s.path(digest))
	if err != nil {
		return nil, fmt.Errorf("localstore pull: %w", err)
	}
	return f, nil
}

func (s *LocalStore) PullToFile(ctx context.Context, digest, path string) error {
	f, err := os.Create(path)
	if err != nil {
		return fmt.Errorf("localstore pulltofile: create: %w", err)
	}
	defer f.Close()
	src, err := s.Pull(ctx, digest)
	if err != nil {
		return err
	}
	defer src.Close()
	if _, err := io.Copy(f, src); err != nil {
		return fmt.Errorf("localstore pulltofile: copy: %w", err)
	}
	return nil
}

// AcquireRef persists a refcount bump so restart does not lose in-flight
// build protection (DESIGN.md §5.1).
func (s *LocalStore) AcquireRef(digest string) (func(), error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	if err := s.bumpRef(digest, +1); err != nil {
		return nil, err
	}
	return func() {
		s.mu.Lock()
		defer s.mu.Unlock()
		_ = s.bumpRef(digest, -1) // best-effort decrement; -1 clamps at 0
	}, nil
}

type refFile struct {
	Count   int       `json:"count"`
	Updated time.Time `json:"updated"`
}

func (s *LocalStore) bumpRef(digest string, d int) error {
	rf := refFile{}
	p := s.refPath(digest)
	if b, err := os.ReadFile(p); err == nil {
		if err := json.Unmarshal(b, &rf); err != nil {
			return fmt.Errorf("localstore refcount read: %w", err)
		}
	}
	rf.Count += d
	if rf.Count < 0 {
		rf.Count = 0
	}
	rf.Updated = time.Now()
	if rf.Count == 0 {
		return os.Remove(p)
	}
	if err := os.MkdirAll(filepath.Dir(p), 0o755); err != nil {
		return fmt.Errorf("localstore refcount mkdir: %w", err)
	}
	b, _ := json.Marshal(rf)
	if err := os.WriteFile(p, b, 0o644); err != nil {
		return fmt.Errorf("localstore refcount write: %w", err)
	}
	return nil
}

func (s *LocalStore) refCount(digest string) int {
	b, err := os.ReadFile(s.refPath(digest))
	if err != nil {
		return 0
	}
	rf := refFile{}
	_ = json.Unmarshal(b, &rf)
	return rf.Count
}

func (s *LocalStore) blobDir() string { return filepath.Join(s.root, "blobs") }

func (s *LocalStore) path(digest string) string {
	sum := digest
	if i := indexByte(digest, ':'); i >= 0 {
		sum = digest[i+1:]
	}
	return filepath.Join(s.blobDir(), sum[:2], sum)
}

func (s *LocalStore) refPath(digest string) string {
	return filepath.Join(s.root, "refs", digest+".json")
}

func indexByte(s string, c byte) int {
	for i := 0; i < len(s); i++ {
		if s[i] == c {
			return i
		}
	}
	return -1
}
```

```go
// GC reclaims blobs older than TTL unless referenced (refcount > 0).
type GC struct {
	store *LocalStore
	ttl   time.Duration
	now   func() time.Time
}

func NewGC(s *LocalStore, ttl time.Duration) *GC {
	return &GC{store: s, ttl: ttl, now: time.Now}
}

func (g *GC) SetNow(fn func() time.Time) { g.now = fn }

func (g *GC) Run(ctx context.Context) error {
	cutoff := g.now().Add(-g.ttl)
	return filepath.WalkDir(g.store.blobDir(), func(path string, d os.DirEntry, err error) error {
		if err != nil || d.IsDir() {
			return err
		}
		info, err := d.Info()
		if err != nil {
			return err
		}
		if info.ModTime().After(cutoff) {
			return nil
		}
		digest := digestFromPath(g.store.blobDir(), path)
		if g.store.refCount(digest) > 0 {
			return nil
		}
		return os.Remove(path)
	})
}
```

```go
// RenderToml renders buildkitd.toml registry sections.
func RenderToml(cfgs []Config) (string, error) {
	sorted := append([]Config(nil), cfgs...)
	sort.Slice(sorted, func(i, j int) bool { return sorted[i].Host < sorted[j].Host })
	var b strings.Builder
	for _, c := range sorted {
		fmt.Fprintf(&b, "[registry.%q]\n", c.Host)
		if c.Insecure {
			b.WriteString("  insecure = true\n")
		}
		if c.HTTP {
			b.WriteString("  http = true\n")
		}
		b.WriteString("\n")
	}
	return b.String(), nil
}
```

```go
// Package creds implements the unified credential table. Credentials are
// resolved caller-side and travel per-Solve over the buildkit session;
// runtime instances hold none (DESIGN.md §7.1).
package creds

import (
	"encoding/base64"
	"errors"
	"os"
	"strings"
)

type Kind string

const (
	Registry Kind = "registry"
	Git      Kind = "git"
)

type AuthRef struct {
	Env string // environment variable holding "username:password"
}

type Credential struct {
	Kind  Kind
	Match string // registry host[/namespace] or git host; longest prefix wins
	Auth  AuthRef
}

type Table []Credential

// Match returns the credential with the longest matching prefix, or nil
// (anonymous) when nothing matches.
func (t Table) Match(kind Kind, host string) *Credential {
	var best *Credential
	for i := range t {
		c := &t[i]
		if c.Kind != kind {
			continue
		}
		if !strings.HasPrefix(host, c.Match) {
			continue
		}
		if best == nil || len(c.Match) > len(best.Match) {
			best = c
		}
	}
	return best
}

type Secret struct {
	Username string
	Secret   string
}

// Resolve reads the env var. The returned secret is only used to build the
// session auth provider; it is never logged or sent with the request body.
func (c *Credential) Resolve() (Secret, error) {
	v, ok := os.LookupEnv(c.Auth.Env)
	if !ok {
		return Secret{}, errors.New("creds: env " + c.Auth.Env + " not set")
	}
	user, pass, ok := strings.Cut(v, ":")
	if !ok {
		// tolerate raw base64 basic auth ("user:pass" encoded)
		raw, err := base64.StdEncoding.DecodeString(v)
		if err != nil {
			return Secret{}, errors.New("creds: env " + c.Auth.Env + " malformed")
		}
		user, pass, ok = strings.Cut(string(raw), ":")
		if !ok {
			return Secret{}, errors.New("creds: env " + c.Auth.Env + " malformed")
		}
	}
	return Secret{Username: user, Secret: pass}, nil
}
```

```go
// Package docker supplies buildkitd runtimes as local docker containers
// (DESIGN.md §4.2: container path is the default; the daemon's built-in
// buildkit is a single-instance zero-startup fallback, not the default).
package docker

import (
	"context"
	"crypto/rand"
	"encoding/hex"
	"fmt"
	"sync"
	"time"

	"github.com/docker/docker/api/types/container"
	"github.com/docker/docker/api/types/mount"
	dclient "github.com/docker/docker/client"
	"github.com/docker/go-connections/nat"

	"github.com/funcx27/kite/modules/buildfleet/registry"
	"github.com/funcx27/kite/modules/buildfleet/resource"
	"github.com/funcx27/kite/modules/buildfleet/runtime"
)

type Config struct {
	Total      resource.Resources // node capacity minus system reserve
	Image      string             // global BuildkitImage (DESIGN.md §5.2)
	Registries registry.Table
	// DockerAPI overrides the client (tests / custom socket).
	DockerAPI dclient.APIClient
}

type Provider struct {
	cfg   Config
	cli   dclient.APIClient
	mu    sync.Mutex
	live  resource.Resources
	ports map[string]string // runtime id -> container id
}

func New(cfg Config) *Provider {
	p := &Provider{cfg: cfg, ports: map[string]string{}}
	p.cli = cfg.DockerAPI
	return p
}

func (p *Provider) Capacity() resource.Resources {
	p.mu.Lock()
	defer p.mu.Unlock()
	return p.cfg.Total.Sub(p.live)
}

func (p *Provider) trackLive(r resource.Resources) {
	p.mu.Lock()
	defer p.mu.Unlock()
	p.live = p.live.Add(r)
}

func (p *Provider) untrackLive(r resource.Resources) {
	p.mu.Lock()
	defer p.mu.Unlock()
	p.live = p.live.Sub(r)
}

// Provision starts a restricted buildkitd container and waits for its gRPC
// health endpoint.
func (p *Provider) Provision(ctx context.Context, spec runtime.Spec) (*runtime.Runtime, error) {
	toml, err := registry.RenderToml(p.cfg.Registries)
	if err != nil {
		return nil, fmt.Errorf("docker provision: render toml: %w", err)
	}

	name := fmt.Sprintf("buildfleet-%s", randID())
	hc := &container.HostConfig{
		Resources: container.Resources{
			NanoCPUs: int64(spec.Resources.CPU * 1e9),
			Memory:   int64(spec.Resources.Memory),
		},
		PortBindings: nat.PortMap{"1234/tcp": []nat.PortBinding{{HostIP: "127.0.0.1"}}},
	}
	cfg := &container.Config{
		Image: p.cfg.Image,
		Cmd:   []string{"buildkitd", "--addr", "tcp://0.0.0.0:1234"},
		Labels: map[string]string{"buildfleet": "runtime"},
	}

	resp, err := p.cli.ContainerCreate(ctx, cfg, hc, nil, nil, name)
	if err != nil {
		return nil, fmt.Errorf("docker provision: create: %w", err)
	}
	p.trackLive(spec.Resources)
	p.mu.Lock()
	p.ports[resp.ID] = resp.ID
	p.mu.Unlock()

	if err := p.cli.ContainerStart(ctx, resp.ID, container.StartOptions{}); err != nil {
		_ = p.Dispose(context.WithoutCancel(ctx), resp.ID)
		return nil, fmt.Errorf("docker provision: start: %w", err)
	}

	// TODO(render toml): config must reach the container — mount as a file
	// via hc.Mounts (mount.TypeVolume with a temp config dir) before start.
	_ = toml

	port, err := p.waitReady(ctx, resp.ID, 60*time.Second)
	if err != nil {
		_ = p.Dispose(context.WithoutCancel(ctx), resp.ID)
		return nil, err
	}
	return &runtime.Runtime{
		ID:    resp.ID,
		Spec:  spec,
		Addr:  "127.0.0.1:" + port,
		State: runtime.Warm,
	}, nil
}

// waitReady inspects the container until its host port is assigned, then
// dials the buildkit gRPC health endpoint (interval 500ms).
func (p *Provider) waitReady(ctx context.Context, cid string, timeout time.Duration) (string, error) {
	deadline := time.Now().Add(timeout)
	for time.Now().Before(deadline) {
		insp, err := p.cli.ContainerInspect(ctx, cid)
		if err == nil && insp.State.Running {
			binds := insp.NetworkSettings.Ports["1234/tcp"]
			if len(binds) > 0 && binds[0].HostPort != "" {
				return binds[0].HostPort, nil
			}
		}
		select {
		case <-ctx.Done():
			return "", ctx.Err()
		case <-time.After(500 * time.Millisecond):
		}
	}
	return "", fmt.Errorf("docker provision: ready timeout for %s", cid)
}

// Dispose is idempotent per the Provider contract.
func (p *Provider) Dispose(ctx context.Context, id string) error {
	err := p.cli.ContainerRemove(ctx, id, container.RemoveOptions{Force: true})
	p.mu.Lock()
	delete(p.ports, id)
	p.mu.Unlock()
	if err != nil && dclient.IsErrNotFound(err) {
		return nil
	}
	return err
}

func randID() string {
	b := make([]byte, 6)
	_, _ = rand.Read(b)
	return hex.EncodeToString(b)
}

var _ runtime.Provider = (*Provider)(nil)
var _ = mount.TypeVolume
```

```go
// Package k8s supplies buildkitd runtimes as rootless pods. Pod shape is
// declarative (BasePod + per-profile overrides) with map-merge / list-append
// / scalar-replace semantics; the image is injected from global config at
// provision time, never from the template (DESIGN.md §5.2).
package k8s

import (
	corev1 "k8s.io/api/core/v1"
)

type PodTemplate struct {
	Annotations       map[string]string
	Labels            map[string]string
	NodeSelector      map[string]string
	Tolerations       []corev1.Toleration
	Affinity          *corev1.Affinity
	Env               []corev1.EnvVar
	Volumes           []corev1.Volume
	SecurityContext   *corev1.PodSecurityContext
	ServiceAccount    string
	PriorityClassName string
}

// Merge applies override over base: maps merge key-wise (override wins),
// lists append, scalars replace when override is non-zero.
func Merge(base, over *PodTemplate) *PodTemplate {
	if base == nil {
		return over
	}
	if over == nil {
		return base
	}
	out := &PodTemplate{
		Annotations:     mergeMap(base.Annotations, over.Annotations),
		Labels:          mergeMap(base.Labels, over.Labels),
		NodeSelector:    mergeMap(base.NodeSelector, over.NodeSelector),
		Tolerations:     append(append([]corev1.Toleration{}, base.Tolerations...), over.Tolerations...),
		Affinity:        firstNonNil(base.Affinity, over.Affinity),
		Env:             append(append([]corev1.EnvVar{}, base.Env...), over.Env...),
		Volumes:         append(append([]corev1.Volume{}, base.Volumes...), over.Volumes...),
		SecurityContext: firstNonNil(base.SecurityContext, over.SecurityContext),
	}
	if over.ServiceAccount != "" {
		out.ServiceAccount = over.ServiceAccount
	} else {
		out.ServiceAccount = base.ServiceAccount
	}
	if over.PriorityClassName != "" {
		out.PriorityClassName = over.PriorityClassName
	} else {
		out.PriorityClassName = base.PriorityClassName
	}
	return out
}

func mergeMap(base, over map[string]string) map[string]string {
	if base == nil && over == nil {
		return nil
	}
	out := map[string]string{}
```

```go
// Package builder drives buildkit Solve against a pooled runtime.
// Credentials attach per-Solve via the session; the runtime instance holds
// none (DESIGN.md §7.1).
package builder

import (
	"context"
	"fmt"

	"github.com/moby/buildkit/client"
	"github.com/moby/buildkit/client/llb"
	"github.com/moby/buildkit/session"
	"github.com/moby/buildkit/session/auth/authprovider"

	"github.com/funcx27/kite/modules/buildfleet/creds"
)

type SolveInput struct {
	ContextDigest string   // ctx://<digest> from the Store
	Dockerfile    string
	Args          map[string]string
	Platforms     []string
	OutputRef     string // image ref to push, or "" for no push
	Creds         creds.Table
}

type Result struct {
	Digest   string
	ImageRef string
}

type Builder struct{}

func (b *Builder) Solve(ctx context.Context, addr string, in SolveInput, status chan *client.SolveStatus) (*Result, error) {
	c, err := client.New(ctx, addr)
	if err != nil {
		return nil, fmt.Errorf("builder: connect %s: %w", addr, err)
	}
	defer c.Close()

	attachables := []session.Attachable{authprovider.NewDockerAgentAuthenticator()}
	if sp, err := newCredsSession(in.Creds); err == nil && sp != nil {
		attachables = append(attachables, sp)
	}

	ch, done := exportChannel(status)
	opts := client.SolveOpt{
		LocalDirs: map[string]string{"context": digestToLocal(in.ContextDigest)},
		FrontendAttrs: map[string]string{
			"filename": in.Dockerfile,
		},
		Exports: pushExports(in.OutputRef),
		Session: attachables,
		ExportProgress: ch,
	}
	for k, v := range in.Args {
		opts.FrontendAttrs["build-arg:"+k] = v
	}
	if len(in.Platforms) > 0 {
		opts.FrontendAttrs["platform"] = joinPlatforms(in.Platforms)
	}

	resp, err := c.Solve(ctx, nil, opts, "", status, nil)
	if err != nil {
		return nil, fmt.Errorf("builder: solve: %w", err)
	}
	<-done
	return &Result{Digest: resp.ExporterResponse["containerimage.digest"], ImageRef: in.OutputRef}, nil
}
```

Git credentials go through the same table with `Kind == creds.Git` (SSH/token via session attachables).

Note on context: MVP wires `LocalDirs` to a path the runtime can see — the Store must be on a shared filesystem with docker runtimes (NFS/local host), which is DESIGN.md §6's LocalStore deployment assumption. `digestToLocal` maps `ctx://<digest>` to the store path; the K8s provider later switches to a git/remote source. Keep this seam small.

`fleet/errors.go` — the three-class taxonomy (DESIGN.md §3):

```go
package fleet

// Class marks an error's retry disposition.
type Class interface{ class() string }

// NodeClass errors retry on a different runtime (<= K attempts):
// runtime unreachable, context pull network failure, registry timeout.
type NodeError struct{ Err error }

func (e NodeError) Error() string { return e.Err.Error() }
func (e NodeError) Unwrap() error { return e.Err }

// ResourceClass errors back off and retry (possibly a lower profile):
// unschedulable, provision failure.
type ResourceError struct{ Err error }

func (e ResourceError) Error() string { return e.Err.Error() }
func (e ResourceError) Unwrap() error { return e.Err }
```

`fleet/fleet.go`:

```go
// Package fleet is the public SDK facade: acquire runtime → solve →
// release, with node-class failover and total-time budget (DESIGN.md §3).
package fleet

import (
	"context"
	"fmt"
)

type Request struct { /* mirrors builder.SolveInput + Resources */ }

type BuildResult struct {
	Digest   string
	ImageRef string
	Attempts int
}

type solver interface {
	Solve(ctx context.Context, addr string, in solveInput) (*solveResult, error)
}

type Fleet struct {
	pool      *runtime.Pool
	solver    solver
	maxRetry  int
}

func New(pool *runtime.Pool, s solver, opts ...Option) *Fleet {
	f := &Fleet{pool: pool, solver: s, maxRetry: 3}
	for _, o := range opts {
		o(f)
	}
	return f
}

// Build runs the retry loop. Every attempt: Acquire → Solve → Release
// (defer, so panics in Solve also return the runtime).
func (f *Fleet) Build(ctx context.Context, req Request) (*BuildResult, error) {
	var lastErr error
	for attempt := 1; attempt <= f.maxRetry; attempt++ {
		lease, err := f.pool.Acquire(ctx, req.spec())
		if err != nil {
			return nil, fmt.Errorf("fleet: acquire: %w", err)
		}
		res, err := f.solver.Solve(ctx, lease.Runtime().Addr, req.input())
		if err == nil {
			lease.Release()
			return &BuildResult{Digest: res.Digest, ImageRef: res.ImageRef, Attempts: attempt}, nil
		}
		if isNodeClass(err) {
			lease.Fail() // node suspect: don't reuse blindly
			lastErr = err
			continue
		}
		lease.Release()
		return nil, err // business/resource: surface
	}
	return nil, fmt.Errorf("fleet: exhausted %d attempts: %w", f.maxRetry, lastErr)
}

func isNodeClass(err error) bool {
	var ne NodeError
	return errors.As(err, &ne)
}
```
